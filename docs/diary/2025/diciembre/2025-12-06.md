def get_valid_transitions_set(self) -> Set[str]:
    """Get set of valid transitions from current state using transitions library API"""
    valid_transitions = set()
    
    # Use transitions library API to get transitions from current state
    if hasattr(self.machine, 'get_transitions'):
        for transition in self.machine.get_transitions(source=self.state):
            valid_transitions.add(transition.trigger)
    
    return valid_transitions
Por qué funciona:

Devuelve todos los transitions desde el estado actual

Acceder a .trigger da el nombre del trigger

    Merge new context with existing context
    Delegates to ContextMergingEngine if available

# Agregar este método a ConversationStateMachine:
def merge_context(self, new_context: dict, strategy: str = "merge") -> dict:
    """
    Merge new context with existing context
    
    Args:
        new_context: Dictionary with new context values
        strategy: "merge" (update existing) or "replace" (clear and set new)
    
    Returns:
        Updated context dictionary
    """
    if strategy == "merge":
        # Merge: update existing context
        self.context.update(new_context)
        return self.context.copy()
    elif strategy == "replace":
        # Replace: clear and set new
        self.context.clear()
        self.context.update(new_context)
        return self.context.copy()
    else:
        raise ValueError(f"Unknown merge strategy: {strategy}. Use 'merge' or 'replace'")

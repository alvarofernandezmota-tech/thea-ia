async def create(self, **kwargs) -> T:
    """Create new record with **kwargs"""
    instance = self.model(**kwargs)
    self.session.add(instance)
    await self.session.flush()
    return instance

19:30 - 20:15 | Fix #5: EventTools Datetime Compatibility
Problema: TypeError: str object cannot be interpreted as an integer en EventTools

Root Cause: EventTools.create_event() esperaba datetime pero recibía str

Solución Implementada:

python
# src/theaia/agents/agenda_agent/tools/event_tools.py v1.3
from typing import Union
from datetime import datetime

async def create_event(
    self,
    title: str,
    start_datetime: Union[datetime, str],  # ✅ Acepta ambos tipos
    ...
) -> Dict[str, Any]:
    # ✅ Conversión automática
    if isinstance(start_datetime, str):
        start_datetime = datetime.fromisoformat(start_datetime.replace("Z", "+00:00"))
